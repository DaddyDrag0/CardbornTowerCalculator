 const timeline=groups.map(group=>{const turnEvent=group.events.find(e=>e.type==='turn'),parsed=turnEvent?parseTurn(turnEvent):null,extras=group.events.filter(e=>e.type!=='turn').map(eventLine).join('');let fight='';if(turnEvent&&parsed){const playerAttacking=turnEvent.team==='Allies';fight=playerAttacking?`<div class="dbg-fight player-attack">${fightCard(turnEvent.card,parsed.aHp,parsed.aMax,parsed.aAtk,'player')}<div class="dbg-vs"><i>→</i></div>${fightCard(parsed.target,parsed.dHp,parsed.dMax,parsed.dAtk,'enemy')}</div>`:`<div class="dbg-fight enemy-attack">${fightCard(parsed.target,parsed.dHp,parsed.dMax,parsed.dAtk,'player')}<div class="dbg-vs"><i>←</i></div>${fightCard(turnEvent.card,parsed.aHp,parsed.aMax,parsed.aAtk,'enemy')}</div>`}else if(turnEvent){fight=`<div class="dbg-fight-simple"><b>${esc(turnEvent.card)}</b><span>${esc(compactText(turnEvent.detail||''))}</span></div>`}return`<section class="dbg-turn"><div class="dbg-turn-head"><b>${group.turn===0?'BATTLE START':`TURN ${group.turn}`}</b>${turnEvent?`<span class="${turnEvent.team==='Allies'?'player':'enemy'}">${side(turnEvent.team)} TURN</span>`:'<span></span>'}<i></i></div>${fight}${extras?`<div class="dbg-interactions">${extras}</div>`:''}</section>`}).join('');
 const dialog=document.createElement('dialog');dialog.className='battle-debug';dialog.innerHTML=`<div class="dbg-shell"><div class="dbg-head"><div><span class="dbg-kicker">BATTLE DEBUG</span><h3>Team ${teamIndex+1} · Run ${runIndex+1}</h3><div class="dbg-sub">${run.death<=state.cap?`Death floor ${fmt(run.death)}`:`Floor cap ${fmt(run.cleared)}`} · ${time(run.seconds)}</div></div><div class="dbg-actions"><button data-dbg-copy>Copy debug</button><button data-dbg-close>Close</button></div></div><div class="dbg-scroll"><div class="dbg-matchup">${matchTeam(d?.initialAllies,'YOUR TEAM','player')}<div class="dbg-match-vs">VS</div>${matchTeam(d?.initialEnemies,'ENEMY TEAM','enemy')}</div><div data-dbg-timeline>${timeline||'<div class="dbg-empty">No battle events were recorded.</div>'}</div></div></div>`;
 document.body.appendChild(dialog);dialog.querySelector('[data-dbg-close]').onclick=()=>{dialog.close();dialog.remove()};dialog.querySelector('[data-dbg-copy]').onclick=async e=>{const text=JSON.stringify({team:teamIndex+1,run:runIndex+1,deathFloor:run.death,cleared:run.cleared,debug:d},null,2);try{await navigator.clipboard.writeText(text);e.currentTarget.textContent='Copied!';setTimeout(()=>{if(e.currentTarget?.isConnected)e.currentTarget.textContent='Copy debug'},900)}catch(_){prompt('Copy battle debug:',text)}};dialog.addEventListener('cancel',()=>dialog.remove());dialog.addEventListener('click',e=>{if(e.target===dialog){dialog.close();dialog.remove()}});dialog.showModal();
}
