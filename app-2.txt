function turnStart(f,t,own,foes,r,ctx){f.turn++;let mult=f.next||1,attacks=Math.max(1,f.attacks||1);f.next=1;if(f._restore){f._restore--;if(!f._restore)f.atk=Math.max(f.atk,f.baseAtk)}if(!on(f))return{mult,attacks,target:t};const n=f.ability.name;if(n==='Focused Aim'&&f.turn===1)mult*=1+p(f,0,.15);if(n==='Peasant’s Bravery'&&f.hp/f.max<p(f,1,.30))mult*=1+p(f,0,.10);if(n==='Backstab'&&f.index===3)mult*=1+p(f,0,.30);if(n==='Flame Wall'&&f.turn<=2&&t)damage(f,t,Math.floor(t.max*p(f,0,.08)),r,ctx,true);if(n==='Dark Pact'){f.hp=Math.max(1,f.hp-Math.floor(f.hp*p(f,0,.10)));f.atk=Math.floor(f.atk*(1+p(f,1,p(f,0,.10))))}if(n==='Holy Strike')heal(f,Math.floor(f.max*p(f,0,.15)));if(n==='Treasure')own.forEach(x=>heal(x,Math.floor(f.max*p(f,0,.05))));if(n==='Reflective Coil'&&t&&r()<.5)t.atk=Math.floor(t.atk*(1-p(f,0,.15)));if(n==='Crystal Slam'&&r()<.5){mult*=1.10;heal(f,Math.floor(f.max*p(f,1,.10)))}if(n==='Weak Growth')buff(f,p(f,0,.05));if(n==='The Coming of the Year')buff(f,.01+.02*(f.turn-1));if(n==='Radiant Burst'){own.forEach(x=>buffHp(x,p(f,0,.07)));alive(foes).forEach(x=>damage(f,x,Math.floor(x.hp*p(f,1,p(f,0,.07))),r,ctx,true))}if(n==='Soulpierce Arrow'&&r()<.15)mult*=3;if(n==='Endless Hunger'&&r()<.40)buff(f,p(f,0,.20));if(n==='Critical Mass'){mult*=3;f.atk=Math.max(1,Math.floor(f.atk*(1-p(f,0,.25))))}if(n==='Demonic Fury'){const q=Math.min(.50,Math.floor((1-f.hp/f.max)/.10)*.10);f.atk=Math.max(f.atk,Math.floor(f.baseAtk*(1+q)))}if(n==='Double or Nothing')mult*=r()<.5?0:2;if(n==='Triple Petal Slash'){attacks=3;mult*=.75}if(n==='Celestial Smite'||n==='Double Strike')attacks=2;if(n==='Unstable Power')mult*=.60+r()*1.90;if(n==='Kings Vision')buff(f,p(f,0,.15));if(n==='Assassin’s Blizzard'){t=alive(foes).sort((a,b)=>a.hp-b.hp)[0]||t;if(t?.frozen)mult*=1+p(f,0,.40)}if(n==='Necrotic Bloom'&&f.turn%2===0){f.shields++;own.forEach(x=>heal(x,Math.floor(x.max*(p(f,2,p(f,1,.08))))))}if(n==='Tainted Vitality'){f.hp-=Math.floor(f.max*p(f,0,.08));f.next+=Math.floor(f.max*p(f,0,.08))/Math.max(1,f.atk)}if(n==='Treacherous Feast'){const e=pick(r,alive(foes));if(e){const before=e.hp;damage(f,e,Math.floor(f.atk*p(f,0,.20)),r,ctx);if(before>0&&e.hp<=0)f.hp=f.max}}if(n==='Frost Breath'&&t){if(r()<.35)t.frozen=1;if(t.frozen)mult*=1+p(f,0,.60)}if(n==='Crystalline Prison'&&f.turn%2===0){const e=pick(r,alive(foes));if(e){e.skip++;damage(f,e,Math.floor(e.max*p(f,0,.15)),r,ctx,true)}}if(n==='King’s Command'&&f.turn%2===0&&t){damage(t,t,Math.floor(t.atk*p(f,0,.65)),r,ctx,true);mult=0;attacks=0}if(n==='Flow State'){f.atk=Math.floor(f.atk*1.20);f.dr=Math.min(.50,f.dr+.10);if(f.turn>=5)attacks=2}return{mult,attacks,target:t}}
function corrReduce(t,a,d){if(!t.flags.corrupted)return d;if(t.corr==='BlightedSkin')d=Math.floor(d*(a.atk>t.atk?.85:.90));if(t.corr==='UnholyResolve'&&t.hp/t.max<.5)d=Math.floor(d*.80);if(t.corr==='VoidArmor'&&!t._void){t._void=true;d=Math.floor(d*.40)}return d}
function incoming(t,a,d,r,ctx,ignore){if(ignore||a?.ability.name==='Perfect Strike')return d;d=corrReduce(t,a,d);if(t._vulTurns)d=Math.floor(d*(1+t._vul));if(t.shieldHp){const q=Math.min(t.shieldHp,d);t.shieldHp-=q;d-=q;if(d<=0)return 0}if(t.shields){t.shields--;return 0}d=Math.floor(d*(1-clamp(t.dr,0,.9)));if(!on(t))return d;const n=t.ability.name;if(n==='Nature’s Guardian'&&!t._block){t._block=true;return 0}if(n==='Resilience'&&t.hp/t.max>p(t,1,.40))d=Math.floor(d*(1-p(t,0,.15)));if(n==='Scurrying Escape'&&r()<p(t,0,.20)){heal(t,Math.floor(t.max*p(t,1,.10)));return 0}if(n==='Eye of the Storm'&&r()<p(t,0,.35))return 0;if(n==='Seasoned Veteran')d=Math.min(d,Math.floor(t.max*p(t,0,.35)));if(n==='Grace of the Divine'&&d<t.max*p(t,0,.25)){heal(t,Math.floor(d*(t.flags.fabled?1.25:1)));return 0}if(n==='Mind Fracture'&&r()<.15)return 0;if(n==='Astral Blessing'&&r()<p(t,0,.40)){ctx.team(t).forEach(x=>buff(x,p(t,1,.10)));return 0}if(n==='Royal Decree: Silence'&&r()<clamp(.30+t.hits*.10,0,.60))return 0;if(n==='Between Worlds'&&(t.hits+1)%2===0)d=Math.floor(d*(1-p(t,0,.70)));if(n==='Arcane Cataclysm'){const q=t._arc||p(t,0,.25),th=t.max*q;if(d<th)return 0;d-=Math.floor(th);t._arc=Math.min(.50,q+.05)}if(n==='Fate Reversal'&&!t._fate){t._fate=true;damage(t,a,Math.floor(d*p(t,0,.35)),r,ctx,true);return 0}if(n==='Frozen Guard'&&r()<p(t,0,.40))a.frozen=1;if(n==='Endless Winter'&&d>=t.hp&&!t._winter){t._winter=true;heal(t,d);return 0}if(n==='Moment of Impact'&&d>=t.hp&&!t._impact){t._impact=true;a.skip++;return Math.max(0,t.hp-1)}if(n==='Refraction')damage(t,a,Math.floor(d*p(t,0,.30)),r,ctx,true);return Math.max(0,d)}
function damage(a,t,d,r,ctx,ignore=false){if(!t||t.dead||d<=0)return 0;const before=t.hp,real=incoming(t,a,Math.floor(d),r,ctx,ignore);t.hp-=real;t.hits++;if(t._vulTurns)t._vulTurns--;if(on(t)&&t.ability.name==='Furnace Heart')ctx.team(t).forEach(x=>x.atk=Math.floor(x.atk*(1+p(t,0,.15))));if(on(a)&&a.ability.name==='Void Devastation')a.stored=Math.min(a.max*3,a.stored+real);return Math.max(0,before-Math.max(0,t.hp))}
function corrPost(f,t,dealt,killed){if(!f.flags.corrupted)return;if(f.corr==='RotTouch'&&t&&!t.dead){t.max=Math.max(1,t.max-Math.floor(dealt*.08));t.hp=Math.min(t.hp,t.max)}if(f.corr==='CursedFangs')heal(f,Math.min(Math.floor(dealt*(f.hp/f.max<.5?.14:.08)),Math.floor(f.max*.15)));if(f.corr==='VampiricCorruption'){heal(f,Math.min(Math.floor(dealt*.12),Math.floor(f.max*.15)));if(killed)f.shieldHp+=Math.floor(f.max*.10)}if(f.corr==='EternalRot'){heal(f,Math.min(Math.floor(dealt*.10),Math.floor(f.max*.15)));if(t&&!t.dead)t.atk=Math.floor(t.atk*.94)}if(f.corr==='SoulHarvester'){heal(f,Math.floor(f.max*.10));if(killed){f.atk=Math.floor(f.atk*1.08);f.shieldHp+=Math.floor(f.max*.12)}}}
function after(f,t,dealt,own,foes,r,ctx,killed){corrPost(f,t,dealt,killed);if(!on(f))return;const n=f.ability.name;if(n==='Lich’s Curse'&&t)t.atk=Math.floor(t.atk*(1-p(f,0,.10)));if(n==='Blinding Dive'&&t&&r()<.20){damage(f,t,Math.floor(f.atk*p(f,0,.40)),r,ctx);t.skip++}if(n==='Vampiric Drain')heal(f,Math.floor(f.atk*p(f,0,.50)));if(n==='Winter’s Grasp'&&t){f.max+=Math.floor(t.max*p(f,0,.07));f.hp+=Math.floor(t.max*p(f,0,.07));f.atk+=Math.floor(t.atk*p(f,0,.07));if(r()<.20)t.frozen=1}if(n==='Festering Edge'&&t){t.max=Math.max(1,t.max-Math.floor(f.atk*p(f,0,.50)));t.hp=Math.min(t.hp,t.max);f.atk=Math.floor(f.atk*(1+p(f,1,.25)))}if(n==='Withering Feast'&&t){t.max=Math.max(1,t.max-dealt);t.hp=Math.min(t.hp,t.max)}if(n==='Between Worlds'){const e=pick(r,alive(foes).filter(x=>x!==t));if(e)damage(f,e,Math.floor(f.atk*p(f,0,.35)),r,ctx)}if(n==='Double Strike'&&f.flags.fabled&&killed)heal(f,Math.floor(f.max*.15));if(n==='Imperial Advance'&&killed){f.shields++;if(f.flags.fabled)alive(own).forEach(x=>buff(x,.05));f.next*=f.flags.fabled?.80:.65}if(n==='Starfall Barrage'&&killed)f.atk=Math.floor(f.atk*1.30);if(n==='Cataclysmic Devour'&&killed&&t){f.max+=Math.floor(t.max*.30);f.hp+=Math.floor(t.max*.30);f.atk+=Math.floor(t.atk*.30)}if(n==='Permafrost'&&killed&&t?.frozen)buff(f,.20)}
function die(f,k,team,foes,r,ctx){if(f.hp>0||f.dead)return false;if(on(f)){const n=f.ability.name;if(n==='Soul Reservoir'&&f.revives<2&&r()<.60){f.revives++;f.hp=Math.floor(f.max*.50);k.skip++;return false}if(n==='Shadow Rebirth'&&!f.revives){f.revives++;f.hp=f.max;damage(f,k,f.max,r,ctx,true);return false}if(n==='Full Moon Transformation'&&!f.revives){f.revives++;buff(f,p(f,0,.15));f.hp=f.max;return false}if(n==='Age of Beasts'&&!f.revives){f.revives++;f.hp=Math.floor(f.max*p(f,1,p(f,0,.40)));return false}if(n==='Curse of Decay'&&!f._curse){f._curse=true;f.hp=1;f.shields++;return false}if(n==='Void Devastation'&&k)damage(f,k,f.stored,r,ctx,true);if(n==='Raise Dead'){const dead=team.filter(x=>x.dead);if(dead.length){const q=pick(r,dead),z=fighter(q.card,q.flags,f.side,team.length);z.hp=Math.floor(z.max*p(f,0,.20));team.push(z)}}if(n==='Goblin Rally'){const g=BY.get('Goblin');if(g)for(let i=0;i<2;i++)team.push(fighter(g,flags(),f.side,team.length))}if(n==='Final Stand')foes.forEach(x=>x.atk=Math.floor(x.atk*(1-p(f,0,.20))))}f.dead=true;f.hp=0;k.kills++;if(on(k)){const n=k.ability.name;if(n==='Savage Fury')k.atk=Math.floor(k.atk*(1+p(k,0,.15)));if(n==='Cataclysmic Devour'){k.max+=Math.floor(f.max*.30);k.hp+=Math.floor(f.max*.30);k.atk+=Math.floor(f.atk*.30)}}return true}
function battle(pt,et,r,cid){const ctx={team:f=>f.side==='p'?pt:et};teamStart(pt,et,cid);teamStart(et,pt,'');let pi=0,ei=0,actions=0,changes=0,turns=0;let pf=pt[0],ef=et[0];enter(pf,ef,pt,et,r,ctx);enter(ef,pf,et,pt,r,ctx);let player=true;while(turns++<700){pi=next(pt,pi);ei=next(et,ei);if(pi<0)return{win:false,actions,changes};if(ei<0)return{win:true,actions,changes};pf=pt[pi];ef=et[ei];const f=player?pf:ef,own=player?pt:et,foes=player?et:pt;let t=player?ef:pf;if(f.frozen){f.frozen--;player=!player;continue}if(f.skip){f.skip--;player=!player;continue}const pre=turnStart(f,t,own,foes,r,ctx);t=pre.target||t;for(let hit=0;hit<pre.attacks;hit++){if(!t||t.dead)t=alive(foes)[0];if(!t)break;let m=pre.mult;if(f.ability.name==='Celestial Smite'&&hit===1)m*=1+p(f,0,.25);if(f.ability.name==='Triple Petal Slash')m*=1;const dealt=damage(f,t,Math.floor(f.atk*m),r,ctx,f.ability.name==='Perfect Strike');actions++;let killed=false;if(t.hp<=0)killed=die(t,f,foes,own,r,ctx);after(f,t,dealt,own,foes,r,ctx,killed);if(killed){if(player){ei=next(et,ei+1);changes++;if(ei>=0){t=et[ei];enter(t,f,et,pt,r,ctx)}}else{pi=next(pt,pi+1);changes++;if(pi>=0){t=pt[pi];enter(t,f,pt,et,r,ctx)}}}if((player&&ei<0)||(!player&&pi<0))break}player=!player}return{win:next(et,Math.max(0,ei))<0,actions,changes}}
