// Void border support.
// Void is a normal stackable border with a 10,000,000x rarity multiplier
// and no special combat behavior.
if(!BORDERS.includes('void'))BORDERS.push('void');

const VOID_baseRarity=rarity;
rarity=function(c,f){
  const base=VOID_baseRarity(c,f);
  return f?.void?base*10000000:base;
};
