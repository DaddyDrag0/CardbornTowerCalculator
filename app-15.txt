// Global Corrupted ability selector. The game allows exactly one equipped ability
// at a time; that one ability applies to every Corrupted card on every player team.
const CORRUPTED_OPTIONS=window.CARDBORN_CORRUPTED_ABILITIES||[];
const CORRUPTED_BY_ID=new Map(CORRUPTED_OPTIONS.map(x=>[x.id,x]));
const CORRUPTED_STORE='cardborn-corrupted-selection-v1';
state.corruptedAbility='';
state.corruptedTier=4;

const CORRUPTED_baseLoad=load;
load=function(){
  CORRUPTED_baseLoad();
  try{
    const saved=JSON.parse(localStorage.getItem(CORRUPTED_STORE)||'null');
    const id=String(saved?.ability||'');
    state.corruptedAbility=CORRUPTED_BY_ID.has(id)?id:'';
    const tier=Number(saved?.tier);
    state.corruptedTier=[1,2,3,4].includes(tier)?tier:(CORRUPTED_BY_ID.get(state.corruptedAbility)?.tier||4);
  }catch(_){state.corruptedAbility='';state.corruptedTier=4}
};

const CORRUPTED_baseSave=save;
save=function(){
  CORRUPTED_baseSave();
  try{localStorage.setItem(CORRUPTED_STORE,JSON.stringify({ability:state.corruptedAbility||'',tier:state.corruptedTier||4}))}catch(_){}
};

function CORRUPTED_triggerLabel(v){return v==='StartOfBattle'?'Battle start':v==='OnEntry'?'On entry':v==='PostAttack'?'After attack':'Passive'}
function CORRUPTED_clearResults(){state.teams.forEach(t=>t.result=null)}

function CORRUPTED_renderMenu(){
  const body=root.querySelector('.simulation-body');if(!body)return;
  const bans=body.querySelector('.depth-ban-box');
  const current=CORRUPTED_BY_ID.get(state.corruptedAbility),tier=Math.max(1,Math.min(4,Number(state.corruptedTier)||4));
  const list=CORRUPTED_OPTIONS.filter(x=>x.tier===tier);
  const box=document.createElement('section');
  box.className='corrupted-ability-box';
  box.innerHTML=`<div class="corrupted-ability-head"><div><span>CORRUPTED ABILITY</span><small>One equipped ability applies to every Corrupted card.</small></div><button type="button" data-corr-none class="${current?'':'on'}">${current?'Unequip':'None equipped'}</button></div><div class="corrupted-current"><span>ACTIVE</span><b>${current?esc(current.name):'None'}</b>${current?`<small>Tier ${current.tier} · ${esc(CORRUPTED_triggerLabel(current.trigger))}</small>`:'<small>Corrupted cards use border stats only.</small>'}</div><div class="corrupted-tier-tabs">${[1,2,3,4].map(n=>`<button type="button" data-corr-tier="${n}" class="${tier===n?'on':''}">Tier ${n}</button>`).join('')}</div><div class="corrupted-options">${list.map(a=>`<button type="button" data-corr-ability="${esc(a.id)}" class="${state.corruptedAbility===a.id?'on':''}"><span><b>${esc(a.name)}</b><i>${esc(CORRUPTED_triggerLabel(a.trigger))}</i></span><small>${esc(a.description)}</small></button>`).join('')}</div>`;
  if(bans)body.insertBefore(box,bans);else body.appendChild(box);
}

const CORRUPTED_baseRender=render;
render=function(){CORRUPTED_baseRender();CORRUPTED_renderMenu()};

root.addEventListener('click',e=>{
  const tier=e.target.closest('[data-corr-tier]');
  if(tier){e.preventDefault();e.stopImmediatePropagation();state.corruptedTier=Math.max(1,Math.min(4,Number(tier.dataset.corrTier)||1));save();render();return}
  const ability=e.target.closest('[data-corr-ability]');
  if(ability){e.preventDefault();e.stopImmediatePropagation();const id=String(ability.dataset.corrAbility||'');if(CORRUPTED_BY_ID.has(id)){state.corruptedAbility=id;state.corruptedTier=CORRUPTED_BY_ID.get(id).tier;CORRUPTED_clearResults();save();render()}return}
  const none=e.target.closest('[data-corr-none]');
  if(none){e.preventDefault();e.stopImmediatePropagation();state.corruptedAbility='';CORRUPTED_clearResults();save();render()}
},true);

// Battle debug/replay must use the same global selection as the parallel workers.
replayFloor=function(t,seed,floor){
  const r=rng32(floorSeed(seed,floor)),pt=makePlayer(t),et=makeEnemy(floor,r);
  return replayBattle(pt,et,r,state.corruptedAbility||'',floor);
};
