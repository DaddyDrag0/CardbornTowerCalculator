// Infinite Tower reward summary.
// Rewards generate on every CLEARED floor. Keep the results simple: use the
// team's median cleared floor, total all generated cards through that floor,
// and show only the Infinite luck bonuses earned by that floor.

function REWARDS_cardsOnFloor(f){
  f=Math.max(1,Math.floor(Number(f)||1));
  if(f<=9)return 6;if(f<=19)return 8;if(f<=34)return 10;if(f<=49)return 12;
  if(f<=74)return 15;if(f<=99)return 18;if(f<=149)return 22;if(f<=189)return 25;
  if(f<=225)return 28;if(f<=325)return 32;if(f<=450)return 36;if(f<=525)return 40;
  if(f<=650)return 45;return 50;
}

const REWARDS_BANDS=[
  [1,9,6],[10,19,8],[20,34,10],[35,49,12],[50,74,15],[75,99,18],
  [100,149,22],[150,189,25],[190,225,28],[226,325,32],[326,450,36],
  [451,525,40],[526,650,45],[651,Infinity,50]
];
function REWARDS_totalCards(floor){
  floor=Math.max(0,Math.floor(Number(floor)||0));
  if(floor<1)return 0;
  let total=0;
  for(const[start,end,count]of REWARDS_BANDS){
    if(floor<start)break;
    const last=Math.min(floor,end);
    if(last>=start)total+=(last-start+1)*count;
    if(floor<=end)break;
  }
  return total;
}
function REWARDS_bonuses(floor){
  floor=Math.max(0,Math.floor(Number(floor)||0));
  return{
    luck:8*Math.floor(floor/5),
    shiny:.4*Math.floor(floor/10),
    awakened:Math.floor(floor/15),
    corrupted:.5*Math.floor(floor/12)
  };
}
function REWARDS_fmt(n){
  n=Number(n)||0;
  if(Math.abs(n)>=1000)return compact(n);
  if(Number.isInteger(n))return n.toLocaleString('en-US');
  return n.toFixed(1).replace(/\.0$/,'');
}
function REWARDS_resultHtml(r){
  if(!r)return'';
  const floor=Math.max(0,Math.floor(Number(r.med)||0));
  const cards=REWARDS_totalCards(floor),perFloor=floor?REWARDS_cardsOnFloor(floor):0,b=REWARDS_bonuses(floor);
  const stat=(label,value,small)=>`<div><span>${label}</span><b>${value}</b>${small?`<small>${small}</small>`:''}</div>`;
  return`<section class="reward-result"><div class="reward-result-head"><div><span>INFINITE REWARDS</span><b>Median floor F${floor}</b></div></div><div class="reward-simple-grid">${stat('Cards generated',REWARDS_fmt(cards),`total through F${floor} · ${perFloor}/floor at F${floor}`)}${stat('Luck',`+${REWARDS_fmt(b.luck)}`,'Infinite bonus')}${stat('Shiny Luck',`+${REWARDS_fmt(b.shiny)}`,'Infinite bonus')}${stat('Awakened Luck',`+${REWARDS_fmt(b.awakened)}`,'Infinite bonus')}${stat('Corrupted Luck',`+${REWARDS_fmt(b.corrupted)}`,'Infinite bonus')}</div></section>`;
}
function REWARDS_decorate(){
  root.querySelectorAll('.reward-settings-box').forEach(x=>x.remove());
  const cards=[...root.querySelectorAll('.result-card')],teams=state.teams.filter(t=>t.result);
  cards.forEach((card,i)=>{
    card.querySelector('.reward-result')?.remove();
    const r=teams[i]?.result;if(r)card.insertAdjacentHTML('beforeend',REWARDS_resultHtml(r));
  });
}

const REWARDS_baseRender=render;
render=function(){REWARDS_baseRender();REWARDS_decorate()};
