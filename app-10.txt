// Live Infinite Dungeon simulation progress + 8-way parallel worker pool.
// Combat, abilities, Fabled logic, On Entry/On Death caps, floor stats, RNG and
// result math are still provided by the exact same engine loaded by the main page.

state.progress=null;

const PROGRESS_WORKERS=8;
const PROGRESS_WORKER_URL='./sim-worker.js?v=23';
const PROGRESS_YIELD_MS=32;
const PROGRESS_sleep=()=>new Promise(resolve=>setTimeout(resolve,0));

function PROGRESS_teamSnapshot(t){
  return{
    cards:(t?.cards||[]).map(s=>({name:s.name,flags:{...flags(),...(s.flags||{})}})),
    corr:t?.corr||''
  };
}

function PROGRESS_percent(p){
  if(!p||!p.totalWork)return 0;
  return Math.max(0,Math.min(100,(p.completed/p.totalWork)*100));
}

function PROGRESS_markup(p){
  const pct=PROGRESS_percent(p),floor=Math.max(1,Math.floor(Number(p.floor)||1));
  return`<div class="sim-progress" data-sim-progress aria-live="polite">
    <div class="sim-progress-head">
      <span>SIMULATING</span>
      <b data-progress-team-name>Team ${p.teamIndex+1} · ${p.workerCount} parallel workers</b>
    </div>
    <div class="sim-progress-values">
      <div><small>CURRENT FLOOR</small><strong data-progress-floor>${floor.toLocaleString()}</strong></div>
      <div><small>RUNS DONE</small><strong data-progress-run>${p.teamCompleted} / ${p.totalRuns}</strong></div>
    </div>
    <div class="sim-progress-track" aria-hidden="true"><i data-progress-fill style="width:${pct.toFixed(2)}%"></i></div>
    <div class="sim-progress-foot"><span data-progress-team>Team ${p.teamPosition} of ${p.teamTotal}</span><span data-progress-pct>${Math.floor(pct)}% complete</span></div>
  </div>`;
}

function PROGRESS_updateDom(){
  const p=state.progress,box=root.querySelector('[data-sim-progress]');
  if(!p||!box)return;
  const pct=PROGRESS_percent(p),floor=Math.max(1,Math.floor(Number(p.floor)||1));
  const floorEl=box.querySelector('[data-progress-floor]'),runEl=box.querySelector('[data-progress-run]'),teamName=box.querySelector('[data-progress-team-name]'),teamEl=box.querySelector('[data-progress-team]'),pctEl=box.querySelector('[data-progress-pct]'),fill=box.querySelector('[data-progress-fill]');
  if(floorEl)floorEl.textContent=floor.toLocaleString();
  if(runEl)runEl.textContent=`${p.teamCompleted} / ${p.totalRuns}`;
  if(teamName)teamName.textContent=`Team ${p.teamIndex+1} · ${p.workerCount} parallel workers`;
  if(teamEl)teamEl.textContent=`Team ${p.teamPosition} of ${p.teamTotal}`;
  if(pctEl)pctEl.textContent=`${Math.floor(pct)}% complete`;
  if(fill)fill.style.width=`${pct.toFixed(2)}%`;
}

// Main-thread fallback used only if Web Workers are unavailable or fail to boot.
function PROGRESS_makeEnemy(floor,r,pool){
  const s=infiniteStats(floor);
  return Array.from({length:4},(_,i)=>fighter(pick(r,pool),flags(),'e',i,s));
}
function PROGRESS_replayFloor(t,seed,floor,pool){
  const r=rng32(floorSeed(seed,floor)),pt=makePlayer(t),et=PROGRESS_makeEnemy(floor,r,pool);
  return replayBattle(pt,et,r,AUTO_CORRUPTED_ABILITY,floor);
}
async function PROGRESS_simulateFallback(t,enemyPool,seed){
  const start=state.start,cap=state.cap;
  let cleared=start-1,actions=0,changes=0,last=[],death=cap,debug=null,lastYield=Date.now();
  for(let floor=start;floor<=cap;floor++){
    if(state.progress)state.progress.floor=floor;
    const r=rng32(floorSeed(seed,floor)),pt=makePlayer(t),et=PROGRESS_makeEnemy(floor,r,enemyPool);
    last=et.slice(0,4).map(x=>x.card.name);
    const b=battle(pt,et,r,AUTO_CORRUPTED_ABILITY);
    actions+=b.actions;changes+=b.changes;
    if(!b.win){
      death=floor;
      debug=PROGRESS_replayFloor(t,seed,floor,enemyPool).debug;
      PROGRESS_updateDom();
      return{seed,cleared,actions,changes,last,death,enemy:infiniteStats(floor),seconds:runSeconds({actions,changes}),debug};
    }
    cleared=floor;
    if(Date.now()-lastYield>=PROGRESS_YIELD_MS){
      PROGRESS_updateDom();
      await PROGRESS_sleep();
      lastYield=Date.now();
    }
  }
  debug=PROGRESS_replayFloor(t,seed,cap,enemyPool).debug;
  PROGRESS_updateDom();
  return{seed,cleared,actions,changes,last,death:cap+1,enemy:infiniteStats(cap),seconds:runSeconds({actions,changes}),debug};
}

function PROGRESS_createWorkers(count){
  if(typeof Worker!=='function')return[];
  const out=[];
  try{
    for(let i=0;i<count;i++)out.push(new Worker(PROGRESS_WORKER_URL));
    return out;
  }catch(err){
    out.forEach(w=>{try{w.terminate()}catch(_){}});
    console.warn('Cardborn parallel workers unavailable; using fallback.',err);
    return[];
  }
}

function PROGRESS_runTeamWorkers(workers,{ti,teamPosition,teamTotal,snapshot,bans,start,cap,speed,base,runsPerTeam}){
  return new Promise((resolve,reject)=>{
    const results=new Array(runsPerTeam),floors=new Map();
    let nextRun=0,finished=0,settled=false;

    const cleanupHandlers=()=>workers.forEach(w=>{w.onmessage=null;w.onerror=null});
    const fail=err=>{
      if(settled)return;
      settled=true;cleanupHandlers();reject(err instanceof Error?err:new Error(String(err)));
    };
    const done=()=>{
      if(settled)return;
      settled=true;cleanupHandlers();resolve(results);
    };
    const updateFloor=()=>{
      if(!state.progress||!floors.size)return;
      // Use the most recently reported live floor. It may reset when a worker starts its next run,
      // which correctly reflects that eight independent runs are moving at the same time.
      const vals=[...floors.values()];
      state.progress.floor=vals[vals.length-1]||state.start;
      PROGRESS_updateDom();
    };
    const dispatch=(w,workerIndex)=>{
      if(nextRun>=runsPerTeam){floors.delete(workerIndex);return}
      const runIndex=nextRun++,seed=(base+ti*0x9e3779b9+runIndex*2654435761)>>>0,jobId=`${ti}:${runIndex}:${seed}`;
      w.__cardbornJob={runIndex,jobId,workerIndex};
      floors.set(workerIndex,start);
      updateFloor();
      w.postMessage({type:'run',jobId,team:snapshot,bans,start,cap,speed,seed});
    };

    workers.forEach((w,workerIndex)=>{
      w.onerror=e=>fail(e?.error||e?.message||'Parallel worker error');
      w.onmessage=e=>{
        const msg=e.data||{};
        if(msg.type==='ready')return;
        if(msg.type==='boot-error'||msg.type==='error'){fail(msg.error||'Parallel worker error');return}
        const job=w.__cardbornJob;
        if(!job||msg.jobId!==job.jobId)return;
        if(msg.type==='progress'){
          floors.delete(workerIndex);floors.set(workerIndex,Math.max(1,Math.floor(Number(msg.floor)||start)));
          updateFloor();
          return;
        }
        if(msg.type==='result'){
          results[job.runIndex]=msg.result;
          finished++;
          if(state.progress){
            state.progress.completed++;
            state.progress.teamCompleted=finished;
            const endFloor=msg.result?.death<=cap?msg.result.death:msg.result?.cleared;
            floors.delete(workerIndex);floors.set(workerIndex,Math.max(1,Math.floor(Number(endFloor)||start)));
            updateFloor();
          }
          if(finished>=runsPerTeam){done();return}
          dispatch(w,workerIndex);
        }
      };
      dispatch(w,workerIndex);
    });
  });
}

async function PROGRESS_runFallback(valid,snapshots,base,runsPerTeam){
  const enemyPool=POOL.filter(c=>!state.bans.includes(c.name));
  state.progress.completed=0;
  for(let teamPos=0;teamPos<valid.length;teamPos++){
    const ti=valid[teamPos],a=[],snapshot=snapshots.get(ti);
    Object.assign(state.progress,{teamIndex:ti,teamPosition:teamPos+1,teamTotal:valid.length,teamCompleted:0,totalRuns:runsPerTeam,workerCount:1,floor:state.start});
    PROGRESS_updateDom();
    for(let i=0;i<runsPerTeam;i++){
      a.push(await PROGRESS_simulateFallback(snapshot,enemyPool,(base+ti*0x9e3779b9+i*2654435761)>>>0));
      state.progress.completed++;
      state.progress.teamCompleted=i+1;
      PROGRESS_updateDom();
      await PROGRESS_sleep();
    }
    state.teams[ti].result=summarize(a);
  }
}

run=async function(indices){
  if(state.running)return;
  const valid=[...new Set((indices||[]).filter(i=>i>=0&&i<state.teams.length&&ready(state.teams[i])))];
  if(!valid.length)return;

  const runsPerTeam=state.runs,base=Date.now()>>>0,totalWork=valid.length*runsPerTeam;
  const snapshots=new Map(valid.map(i=>[i,PROGRESS_teamSnapshot(state.teams[i])]));
  const workerCount=Math.min(PROGRESS_WORKERS,runsPerTeam);
  const workers=PROGRESS_createWorkers(workerCount);

  state.running=true;
  state.pAbilities=true;
  state.eAbilities=true;
  state.progress={teamIndex:valid[0],teamPosition:1,teamTotal:valid.length,teamCompleted:0,totalRuns:runsPerTeam,workerCount:workers.length||1,floor:state.start,completed:0,totalWork};
  render();
  await PROGRESS_sleep();

  try{
    if(!workers.length){
      await PROGRESS_runFallback(valid,snapshots,base,runsPerTeam);
    }else{
      for(let teamPos=0;teamPos<valid.length;teamPos++){
        const ti=valid[teamPos],snapshot=snapshots.get(ti);
        Object.assign(state.progress,{teamIndex:ti,teamPosition:teamPos+1,teamTotal:valid.length,teamCompleted:0,totalRuns:runsPerTeam,workerCount:workers.length,floor:state.start});
        PROGRESS_updateDom();
        const results=await PROGRESS_runTeamWorkers(workers,{ti,teamPosition:teamPos+1,teamTotal:valid.length,snapshot,bans:[...state.bans],start:state.start,cap:state.cap,speed:state.speed,base,runsPerTeam});
        state.teams[ti].result=summarize(results);
        await PROGRESS_sleep();
      }
    }
  }catch(err){
    console.warn('Cardborn parallel simulation failed; retrying on main thread.',err);
    workers.forEach(w=>{try{w.terminate()}catch(_){}});
    state.progress.completed=0;
    valid.forEach(i=>state.teams[i].result=null);
    await PROGRESS_runFallback(valid,snapshots,base,runsPerTeam);
  }finally{
    workers.forEach(w=>{try{w.terminate()}catch(_){}});
    state.running=false;
    state.progress=null;
    save();
    render();
  }
};

const PROGRESS_baseRender=render;
render=function(){
  PROGRESS_baseRender();

  // Remove the old result-header "50 RUNS" style badge.
  root.querySelectorAll('.result-head .verified').forEach(el=>el.remove());

  root.classList.toggle('simulation-running',!!state.running);
  if(state.running&&state.progress){
    const actions=root.querySelector('.sim-actions');
    if(actions)actions.innerHTML=PROGRESS_markup(state.progress);
  }
};
