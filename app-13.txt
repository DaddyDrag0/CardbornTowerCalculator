// Parallel-worker stability patch.
// Bulk simulation uses the lightweight sim-worker directly. The worker skips battle
// debug replay during the run, so a long/stalled battle cannot collapse the 8-way
// pool into the one-thread fallback.

PROGRESS_createWorkers=function(count){
  if(typeof Worker!=='function')return[];
  const out=[];
  try{
    for(let i=0;i<count;i++)out.push(new Worker('./sim-worker.js?v=30'));
    return out;
  }catch(err){
    out.forEach(w=>{try{w.terminate()}catch(_){}});
    console.warn('Cardborn parallel workers unavailable; using fallback.',err);
    return[];
  }
};
