// Depth-style effect-first battle tracing.
// Debug only: this does not change combat results. Ability log entries show what
// actually happened in that battle instead of explaining the ability description.

const ABILITY_TRACE_baseDbgSnap=dbgSnap;
dbgSnap=function(f){
  const s=ABILITY_TRACE_baseDbgSnap(f);
  if(!s||!f)return s;
  return{
    ...s,
    dr:Number(f.dr)||0,
    next:Number.isFinite(Number(f.next))?Number(f.next):1,
    attacks:Math.max(0,Math.floor(Number(f.attacks)||0)),
    revives:Math.max(0,Math.floor(Number(f.revives)||0)),
    stored:Number(f.stored)||0,
    silenced:!!f.silenced,
    marked:!!f._marked,
    vulnerability:Number(f._vul)||0,
    vulnerabilityTurns:Math.max(0,Math.floor(Number(f._vulTurns)||0)),
    restoreTurns:Math.max(0,Math.floor(Number(f._restore)||0)),
    voidArmorUsed:!!f._voidArmorUsed,
    plagueBearerTriggered:!!f._plagueBearerTriggered,
    corr:String(f.corr||'')
  };
};

const ABILITY_TRACE_baseUnitDiff=unitDiff;
unitDiff=function(a,b,label){
  const out=ABILITY_TRACE_baseUnitDiff(a,b,label),tag=label?label+' ':'';
  if(!a||!b)return out;
  const pct=n=>`${Math.round((Number(n)||0)*1000)/10}%`;
  const mult=n=>`x${Math.round((Number(n)||0)*100)/100}`;
  if(a.ability!==b.ability)out.push(`${tag}ability ${a.ability||'none'} → ${b.ability||'none'}`);
  if(Math.abs((Number(a.dr)||0)-(Number(b.dr)||0))>1e-9)out.push(`${tag}damage reduction ${pct(a.dr)} → ${pct(b.dr)}`);
  if(Math.abs((Number(a.next)||1)-(Number(b.next)||1))>1e-9)out.push(`${tag}next-hit multiplier ${mult(a.next)} → ${mult(b.next)}`);
  if(a.attacks!==b.attacks)out.push(`${tag}attacks ${a.attacks} → ${b.attacks}`);
  if(a.revives!==b.revives)out.push(`${tag}revive counter ${a.revives} → ${b.revives}`);
  if(a.stored!==b.stored)out.push(`${tag}stored power ${compact(a.stored)} → ${compact(b.stored)}`);
  if(a.silenced!==b.silenced)out.push(`${tag}${b.silenced?'silenced':'silence removed'}`);
  if(a.marked!==b.marked)out.push(`${tag}${b.marked?'marked':'mark removed'}`);
  if(Math.abs((Number(a.vulnerability)||0)-(Number(b.vulnerability)||0))>1e-9)out.push(`${tag}vulnerability ${pct(a.vulnerability)} → ${pct(b.vulnerability)}`);
  if(a.vulnerabilityTurns!==b.vulnerabilityTurns)out.push(`${tag}vulnerability turns ${a.vulnerabilityTurns} → ${b.vulnerabilityTurns}`);
  if(a.restoreTurns!==b.restoreTurns)out.push(`${tag}restore timer ${a.restoreTurns} → ${b.restoreTurns}`);
  if(!a.voidArmorUsed&&b.voidArmorUsed)out.push(`${tag}Void Armor first-hit protection consumed`);
  if(!a.plagueBearerTriggered&&b.plagueBearerTriggered)out.push(`${tag}Plague Bearer threshold triggered`);
  return out;
};

const ABILITY_TRACE_CORRUPTED=new Map((window.CARDBORN_CORRUPTED_ABILITIES||[]).map(x=>[String(x.id||''),x]));
function ABILITY_TRACE_clean(s){return String(s||'').replace(/\s+/g,' ').trim()}
function ABILITY_TRACE_findSource(event,f,pt,et){
  if(event?.card==='Battle Start')return null;
  const all=[...(pt||[]),...(et||[])],same=all.filter(x=>FULL_teamName(x)===event?.team);
  return same.find(x=>x.card?.name===event?.card)||same.find(x=>x.ability?.name===event?.card)||all.find(x=>x.card?.name===event?.card)||all.find(x=>x.ability?.name===event?.card)||f||null;
}
function ABILITY_TRACE_resultText(detail,source,phase){
  let s=ABILITY_TRACE_clean(detail),name=source?.ability?.name||'';
  const prefixes=[
    `${name} + Corrupted effect · ${phase}`,
    `${name} · ${phase}`,
    `Defense modifiers · ${phase}`,
    `Team effects · Battle start`
  ].filter(Boolean);
  for(const pfx of prefixes)if(s.startsWith(pfx)){s=s.slice(pfx.length);break}
  s=s.replace(/^\s*·\s*/,'');
  if(source?.ability?.name==='Haunting Presence')s=s.replace(/^.*?passive aura\s*·\s*Start of turn\s*·\s*/i,'');
  return s||'Triggered; no direct stat or status change.';
}
function ABILITY_TRACE_corruptedName(f){
  if(!f?.flags?.corrupted)return'';
  const c=ABILITY_TRACE_CORRUPTED.get(String(f.corr||''));
  return ABILITY_TRACE_clean(c?.name||c?.id||'');
}
function ABILITY_TRACE_startNames(f){
  const own=FULL_team(f,FULL_CONTEXT)||[],names=[];
  for(const x of FULL_alive(own)){
    if(typeof START_ABILITIES!=='undefined'&&START_ABILITIES.has(x.ability?.name)&&!names.includes(x.ability.name))names.push(x.ability.name);
  }
  for(const x of own){const c=ABILITY_TRACE_corruptedName(x);if(c&&!names.includes(c))names.push(c)}
  return names;
}

const ABILITY_TRACE_baseLogAbility=FULL_logAbility;
FULL_logAbility=function(events,turn,f,phase,before,pt,et,extra=[]){
  const start=events?.length||0,out=ABILITY_TRACE_baseLogAbility(events,turn,f,phase,before,pt,et,extra);
  if(!events||events.length<=start)return out;
  for(let i=start;i<events.length;i++){
    const event=events[i];if(!event||event.type!=='ability')continue;
    try{Object.defineProperty(event,'_abilityTraceDetailed',{value:true,enumerable:false})}catch(_){}

    if(event.card==='Battle Start'){
      const result=ABILITY_TRACE_resultText(event.detail,f,'Battle start'),names=ABILITY_TRACE_startNames(f);
      event.detail=`${names.length?names.join(' + '):'Battle start effects'} · ${result}`;
      continue;
    }

    const source=ABILITY_TRACE_findSource(event,f,pt,et),result=ABILITY_TRACE_resultText(event.detail,source,phase);

    // Generic incoming-damage changes can come from shields, passives, border effects,
    // or several stacked modifiers. Show the exact effect without falsely naming one.
    if(phase==='Defense interaction'){
      event.detail=`Defense interaction · ${result}`;
      continue;
    }

    const abilityName=ABILITY_TRACE_clean(source?.ability?.name||event.card||'Ability'),corrName=phase==='On entry'?ABILITY_TRACE_corruptedName(source):'';
    event.detail=`${abilityName}${corrName?` + ${corrName}`:''} · ${result}`;
  }
  return out;
};
